# Project-level configuration shared by the GTK3 and GTK4 runners.
cmake_minimum_required(VERSION 3.13)
project(runner LANGUAGES CXX)
set(BINARY_NAME "pcalc_express")
set(APPLICATION_ID "ca.richyoung.pcalcexpress")

# The environment selects a one-shot build; the ignored file is a local default.
if(NOT "$ENV{FLUTTER_LINUX_GTK}" STREQUAL "")
  set(LINUX_GTK_VARIANT "$ENV{FLUTTER_LINUX_GTK}")
elseif(NOT DEFINED LINUX_GTK_VARIANT)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.gtk_variant.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/.gtk_variant.cmake")
  else()
    set(LINUX_GTK_VARIANT gtk3)
  endif()
endif()

if(LINUX_GTK_VARIANT STREQUAL "gtk4")
  set(LINUX_GTK_PKG gtk4)
  set(LINUX_GTK_DEFINE FLUTTER_LINUX_GTK4)
elseif(LINUX_GTK_VARIANT STREQUAL "gtk3")
  set(LINUX_GTK_PKG gtk+-3.0)
  set(LINUX_GTK_DEFINE FLUTTER_LINUX_GTK3)
else()
  message(FATAL_ERROR "Unsupported LINUX_GTK_VARIANT: ${LINUX_GTK_VARIANT}. Expected gtk3 or gtk4.")
endif()

message(STATUS "Configuring Linux runner for ${LINUX_GTK_VARIANT} (${LINUX_GTK_PKG})")
cmake_policy(SET CMP0063 NEW)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/linux_project_common.cmake")
